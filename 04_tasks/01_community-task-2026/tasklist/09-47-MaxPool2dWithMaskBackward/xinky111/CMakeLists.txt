cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(max_pool_backward LANGUAGES ASC CXX)
set(POOL_ARCH "dav-2201" CACHE STRING "Ascend architecture")
add_library(max_pool_backward SHARED launch.asc)
target_include_directories(max_pool_backward PRIVATE op_kernel)
target_compile_options(max_pool_backward PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${POOL_ARCH}>)
option(POOL_EXACT_OUTPUT "Use exact-size output and architecture-specific tail" OFF)
if(POOL_EXACT_OUTPUT)
    target_compile_definitions(max_pool_backward PRIVATE POOL_EXACT_OUTPUT=1)
endif()
if(POOL_ARCH STREQUAL "dav-2201")
    target_compile_definitions(max_pool_backward PRIVATE POOL_SUPPORT_BF16=1)
endif()
